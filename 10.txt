#!/bin/bash
echo "Digite o nome do usuário:"
read usuario
linha=$(grep "^$usuario:" /etc/passwd)
if [ -z "$linha" ]; then
  echo "O usuário $usuario não existe no sistema."
else
  home=$(echo "$linha" | cut -d: -f6)
  echo "Nome do usuário: $usuario"
  echo "Diretório de trabalho: $home"
  echo "Espaço utilizado: $(du -sh "$home" 2>/dev/null | cut -f1)"
fi
