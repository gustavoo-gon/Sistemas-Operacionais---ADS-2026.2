#!/bin/bash
echo "1 - Procurar um arquivo utilizando find"
echo "2 - Procurar um comando utilizando whereis"
echo "3 - Procurar um arquivo utilizando locate"
echo "Escolha uma opção:"
read op
if [ "$op" = "1" ]; then
  echo "Digite o nome do arquivo:"
  read nome
  find / -name "$nome" 2>/dev/null
elif [ "$op" = "2" ]; then
  echo "Digite o nome do comando:"
  read nome
  whereis "$nome"
elif [ "$op" = "3" ]; then
  echo "Digite o nome do arquivo:"
  read nome
  locate "$nome"
else
  echo "Opção inválida."
fi
