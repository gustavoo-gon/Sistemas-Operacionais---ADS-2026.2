#!/bin/bash
op=0
while [ "$op" != "4" ]; do
  echo ""
  echo "===== GERENCIAMENTO DE PROCESSOS ====="
  echo "1 - Listar processos"
  echo "2 - Procurar processo"
  echo "3 - Encerrar processo"
  echo "4 - Sair"
  echo "Escolha uma opção:"
  read op
  if [ "$op" = "1" ]; then
    ps aux
  elif [ "$op" = "2" ]; then
    echo "Digite o nome do processo:"
    read proc
    resultado=$(ps aux | grep "$proc" | grep -v grep)
    if [ -z "$resultado" ]; then
      echo "Nenhum processo encontrado."
    else
      echo "$resultado"
    fi
  elif [ "$op" = "3" ]; then
    echo "Digite o PID do processo:"
    read pid
    ps -p $pid > /dev/null 2>&1
    if [ $? -ne 0 ]; then
      echo "PID não encontrado."
    else
      echo "Deseja realmente encerrar o processo $pid? (s/n)"
      read resp
      if [ "$resp" = "s" ]; then
        kill $pid
        echo "Processo encerrado."
      else
        echo "Operação cancelada."
      fi
    fi
  elif [ "$op" = "4" ]; then
    echo "Saindo..."
  else
    echo "Opção inválida."
  fi
done
