#!/bin/bash

clear

read -p "Informe o n1: " n1
read -p "Informe o n1: " n2

echo "MENU: "

read -p "1=SOMA | 2=SUBTRAÇÃO | 3=MULT | 4=DIV | 5=SAIR: " c

case $c in

    1)
        soma=$((n1+n2))
        echo "a soma de $n1+$n2 é : $soma"
        ;;

    2)
       sub=$((n1-n2))
       echo "a subtração de $n1-$n2 é : $sub" 
       ;;
    3) 
        mult=$((n1*n2))
       echo "a multiplicação de $n1*$n2 é : $mult"
       ;;
    4)
        div=$((n1/n2))
       echo "a divisão de $n1/$n2 é : $div"
       ;;

    5)
       echo "VOCE SAIU DO PROGRAMA"
    *)
        echo "A OPÇÃO DIGITADA NÃO EXISTE"
        ;;
esac
