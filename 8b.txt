#!/bin/bash
echo "Digite o nome do primeiro arquivo:"
read arq1
echo "Digite o nome do segundo arquivo:"
read arq2
echo "Digite o nome do arquivo de destino:"
read destino
if [ -f "$arq1" ] && [ -f "$arq2" ]; then
  cat "$arq1" >> "$destino"
  cat "$arq2" >> "$destino"
  echo "Conteúdo adicionado ao arquivo $destino."
else
  echo "Um dos arquivos não existe."
fi
