#!/bin/bash
op=0
while [ "$op" != "7" ]; do
  echo ""
  echo "===== GERENCIAMENTO DE ARQUIVOS ====="
  echo "1 - Procurar arquivo"
  echo "2 - Procurar palavra em arquivo"
  echo "3 - Renomear arquivo"
  echo "4 - Mover arquivo"
  echo "5 - Remover arquivo"
  echo "6 - Comparar arquivos"
  echo "7 - Sair"
  echo "Escolha uma opção:"
  read op
  if [ "$op" = "1" ]; then
    echo "Digite o nome do arquivo:"
    read nome
    find / -name "$nome" 2>/dev/null
  elif [ "$op" = "2" ]; then
    echo "Digite a palavra:"
    read palavra
    echo "Digite o nome do arquivo:"
    read arq
    if [ -f "$arq" ]; then
      grep -n "$palavra" "$arq"
      if [ $? -ne 0 ]; then
        echo "Palavra não encontrada."
      fi
    else
      echo "O arquivo não existe."
    fi
  elif [ "$op" = "3" ]; then
    echo "Digite o nome atual do arquivo:"
    read atual
    echo "Digite o novo nome:"
    read novo
    if [ -f "$atual" ]; then
      mv "$atual" "$novo"
      echo "Arquivo renomeado."
    else
      echo "O arquivo não existe."
    fi
  elif [ "$op" = "4" ]; then
    echo "Digite o nome do arquivo:"
    read arq
    echo "Digite o diretório de destino:"
    read dir
    if [ -f "$arq" ] && [ -d "$dir" ]; then
      mv "$arq" "$dir"
      echo "Arquivo movido."
    else
      echo "Arquivo ou diretório não encontrado."
    fi
  elif [ "$op" = "5" ]; then
    echo "Digite o nome do arquivo:"
    read arq
    if [ -f "$arq" ]; then
      rm "$arq"
      echo "Arquivo removido."
    else
      echo "O arquivo não foi encontrado."
    fi
  elif [ "$op" = "6" ]; then
    echo "Digite o nome do primeiro arquivo:"
    read arq1
    echo "Digite o nome do segundo arquivo:"
    read arq2
    if [ -f "$arq1" ] && [ -f "$arq2" ]; then
      diff "$arq1" "$arq2" > /dev/null
      if [ $? -eq 0 ]; then
        echo "Os arquivos são iguais."
      else
        echo "Os arquivos são diferentes:"
        diff "$arq1" "$arq2"
      fi
    else
      echo "Um dos arquivos não existe."
    fi
  elif [ "$op" = "7" ]; then
    echo "Saindo..."
  else
    echo "Opção inválida."
  fi
done
