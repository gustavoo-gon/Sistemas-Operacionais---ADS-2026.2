#!/bin/bash
echo "Digite o PID do processo:"
read pid
ps -p $pid > /dev/null 2>&1
if [ $? -ne 0 ]; then
  echo "O PID $pid não foi encontrado na lista de processos."
else
  echo "PID informado: $pid"
  echo "Deseja realmente encerrar esse processo? (s/n)"
  read resp
  if [ "$resp" = "s" ]; then
    kill $pid
    echo "Comando kill enviado para o processo $pid."
  else
    echo "Operação cancelada."
  fi
fi
